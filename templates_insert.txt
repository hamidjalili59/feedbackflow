  static List<FormTemplatePreset> presetsForLanguage(String languageCode) {
    return switch (languageCode) {
      'en' => _englishPresets,
      'zh' => _chinesePresets,
      _ => presets,
    };
  }

  static FormTemplatePreset byType(FormTemplateType type, {String languageCode = 'fa'}) =>
      presetsForLanguage(languageCode).firstWhere((preset) => preset.type == type);
